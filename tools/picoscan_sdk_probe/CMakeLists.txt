cmake_minimum_required(VERSION 3.22)
project(picoscan_sdk_probe LANGUAGES CXX)
find_package(sick_perception_sdk REQUIRED COMPONENTS compact_format)
add_executable(picoscan_sdk_probe main.cpp)
target_compile_features(picoscan_sdk_probe PRIVATE cxx_std_17)
target_link_libraries(picoscan_sdk_probe PRIVATE sick_perception_sdk::compact_format)
if(WIN32)
  target_link_libraries(picoscan_sdk_probe PRIVATE ws2_32)
endif()
